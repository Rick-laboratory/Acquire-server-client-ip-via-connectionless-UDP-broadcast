cmake_minimum_required(VERSION 3.16)
project(UdpPeerDiscovery LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This Winsock example requires Windows.")
endif()
find_package(Threads REQUIRED)
add_library(discovery INTERFACE)
target_include_directories(discovery INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(discovery INTERFACE cxx_std_17)
target_compile_definitions(discovery INTERFACE _WIN32_WINNT=0x0600)
target_link_libraries(discovery INTERFACE ws2_32)
add_executable(discovery-client Inline_IPv4_Broadcast_UDP/main.cpp)
add_executable(discovery-server Inline_IPv4_Receiver/main.cpp)
target_link_libraries(discovery-client PRIVATE discovery)
target_link_libraries(discovery-server PRIVATE discovery)
include(CTest)
if(BUILD_TESTING)
    add_executable(discovery-test tests/discovery_test.cpp)
    target_link_libraries(discovery-test PRIVATE discovery Threads::Threads)
    add_test(NAME loopback-discovery COMMAND discovery-test)
    set_tests_properties(loopback-discovery PROPERTIES TIMEOUT 15)
endif()
foreach(target discovery-client discovery-server discovery-test)
    if(TARGET ${target})
        set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
        if(MSVC)
            target_compile_options(${target} PRIVATE /W4 /permissive-)
        else()
            target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endforeach()
